export default function App() {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  const navigate = useNavigate();

  useEffect(() => {
    let active = true;

    async function restoreSession() {
      if (!getToken()) {
        if (active) {
          setLoading(false);
        }
        return;
      }

      try {
        const result = await getCurrentUser();

        if (active) {
          setUser(result?.data?.user ?? null);
        }
      } catch {
        clearToken();

        if (active) {
          setUser(null);
        }
      } finally {
        if (active) {
          setLoading(false);
        }
      }
    }

    restoreSession();

    return () => {
      active = false;
    };
  }, []);

  function handleLogin(authenticatedUser) {
    setUser(authenticatedUser);
    navigate('/', { replace: true });
  }

  function handleLogout() {
    clearToken();
    setUser(null);
    navigate('/login', { replace: true });
  }

  return (
    <Routes>
      <Route
        path="/login"
        element={
          loading ? (
            <LoadingScreen />
          ) : (
            <LoginPage user={user} onLogin={handleLogin} />
          )
        }
      />

      <Route
        element={
          <ProtectedWorkspace
            user={user}
            loading={loading}
            onLogout={handleLogout}
          />
        }
      >
        <Route index element={<DashboardPage />} />

        {pages.map((page) => (
          <Route
            key={page.path}
            path={page.path}
            element={
              page.path === 'users' ? (
                <UsersPage user={user} />
              ) : page.path === 'cycles' ? (
                <CyclesPage user={user} />

                <TasksPage user={user} />

function daysUntil(value) {
  const end = parseDate(value);
  if (!end) return null;
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  return Math.round((end - today) / 86400000);
}

function describeDeadline(cycle) {
  const days = daysUntil(cycle.endDate);
  if (days === null) return '';
  if (days > 0) return `closes ${formatDate(cycle.endDate)} (${days} day${days === 1 ? '' : 's'} left)`;
  if (days === 0) return `scheduled end date is today (${formatDate(cycle.endDate)})`;
  return `scheduled end date passed ${-days} day${days === -1 ? '' : 's'} ago`;
}

// Same rule as the server: N/A is excluded from the mean; weighted = average * weight / 5.
function computeScore(criteria, scores, weight, scaleMax) {
  let sum = 0;
  let counted = 0;
  let na = 0;
  let answered = 0;

  criteria.forEach((criterion) => {
    const value = scores[criterion.id];
    if (value === undefined || value === null) return;
    answered += 1;
    if (value === 'NA') {
      na += 1;
    } else {
      sum += Number(value);
      counted += 1;
    }
  });

  const average = counted ? sum / counted : null;
  return {
    answered,
    na,
    counted,
    average,
    weighted: average === null ? null : (average * weight) / 5,
    percent: average === null ? null : Math.max(0, Math.min(100, (average / (scaleMax || 5)) * 100)),
  };
}

function labelFor(version, value) {
  try {
    let labels = version?.scaleLabels;
    if (typeof labels === 'string') labels = JSON.parse(labels);
    if (!labels) return '';
    const found = Array.isArray(labels)
      ? labels[value - version.scaleMin]
      : labels[value] ?? labels[String(value)];
    if (found && typeof found === 'object') return String(found.label ?? found.text ?? '');
    return found ? String(found) : '';
  } catch {
    return '';
  }
}

function matchesFilter(row, filter) {
  const status = effectiveStatus(row);
  if (filter === 'all') return true;
  if (filter === 'returned') return status === 'returned' || status === 'rejected';
  return status === filter;
}

const API_BASE_URL =
  import.meta.env.VITE_API_BASE_URL || 'http://localhost:5000/api';

const TOKEN_KEY = 'asems_token';

export function getToken() {
  return sessionStorage.getItem(TOKEN_KEY);
}

export function saveToken(token) {
  sessionStorage.setItem(TOKEN_KEY, token);
}

export function clearToken() {
  sessionStorage.removeItem(TOKEN_KEY);
}

export async function apiRequest(path, options = {}) {
  const token = getToken();

  const headers = {
    'Content-Type': 'application/json',
    ...(token ? { Authorization: `Bearer ${token}` } : {}),
    ...(options.headers || {}),
  };

  const response = await fetch(`${API_BASE_URL}${path}`, {
    ...options,
    headers,
  });

  const payload = await response.json().catch(() => null);

  if (!response.ok || payload?.success === false) {
    const message =
      payload?.error?.message ||
      payload?.message ||
      `Request failed (${response.status})`;

    if (response.status === 401) {
      clearToken();
    }

    throw new Error(message);
  }

  return payload;
}

export async function loginRequest(email, password) {
  const payload = await apiRequest('/auth/login', {
    method: 'POST',
    body: JSON.stringify({ email, password }),
  });

  const token = payload?.data?.token;

  if (!token) {
    throw new Error('Login succeeded, but the server did not return a token.');
  }

  saveToken(token);
  return payload;
}

export async function getCurrentUser() {
  return apiRequest('/auth/me');
}

export async function logoutRequest() {
  clearToken();
}

export async function getApiHealth() {
  return apiRequest('/health');
}

  try {
    const [rows] = await connection.query(`
      SELECT
        1 AS connection_ok,
        DATABASE() AS database_name,
        CURRENT_TIMESTAMP AS database_time
    `);

import jwt from 'jsonwebtoken';

import { env } from '../config/env.js';
import { pool } from '../config/database.js';

// Verify the token and reload the user from the database.
export async function authenticate(req, res, next) {
  const authorization = req.get('authorization');

  if (!authorization?.startsWith('Bearer ')) {
    return res.status(401).json({
      success: false,
      message: 'Authentication is required.',
      code: 'AUTHENTICATION_REQUIRED',
    });
  }

  const token = authorization.slice(7).trim();

  if (!token) {
    return res.status(401).json({
      success: false,
      message: 'Authentication is required.',
      code: 'AUTHENTICATION_REQUIRED',
    });
  }

  let payload;

  try {
    payload = jwt.verify(token, env.JWT_SECRET, {
      issuer: 'asems-api',
      audience: 'asems-web',
    });
  } catch {
    return res.status(401).json({
      success: false,
      message: 'Your session is invalid or has expired. Please sign in again.',
      code: 'INVALID_OR_EXPIRED_TOKEN',
    });
  }

  const userId = Number(payload.sub);

  if (!Number.isSafeInteger(userId) || userId < 1) {
    return res.status(401).json({
      success: false,
      message: 'Invalid authentication token.',
      code: 'INVALID_TOKEN_SUBJECT',
    });
  }

  try {
    const [rows] = await pool.execute(
      `SELECT id, name, email, role, status
         FROM users
        WHERE id = ?
        LIMIT 1`,
      [userId],
    );

    const user = rows[0];

    if (!user || user.status !== 'active') {
      return res.status(401).json({
        success: false,
        message: 'Your account is unavailable.',
        code: 'ACCOUNT_UNAVAILABLE',
      });
    }

    req.user = {
      id: user.id,
      name: user.name,
      email: user.email,
      role: user.role,
      status: user.status,
    };

    return next();
  } catch (error) {
    return next(error);
  }
}

  'admin',
  'student',
  'teacher',

export function authorize(...allowedRoles) {

    throw new Error(
      'Authorization requires at least one allowed role.',
    );
  }

  const invalidRoles = allowedRoles.filter(
    (role) => !VALID_ROLES.has(role),
  );

  if (invalidRoles.length > 0) {
    throw new Error(
      `Invalid authorization role(s): ${invalidRoles.join(', ')}`,

  return function roleAuthorization(req, res, next) {
    if (!req.user) {
      return res.status(401).json({
        success: false,
        message: 'Authentication is required.',
        code: 'AUTHENTICATION_REQUIRED',
      });
    }

    if (!allowedRoles.includes(req.user.role)) {
      return res.status(403).json({
        success: false,
        message: 'You do not have permission to perform this action.',
        code: 'FORBIDDEN',
      });
    }

    return next();
  };
}

export function errorHandler(error, req, res, next) {
  if (res.headersSent) {
    return next(error);
  }

  const statusCode = Number.isInteger(error.statusCode)
    ? error.statusCode
    : error.name === 'ZodError'
      ? 400
      : error.type === 'entity.parse.failed'
        ? 400
        : 500;

  const safeClientErrors = {
    400: 'The request is invalid.',
    401: 'Authentication is required.',
    403: 'You are not permitted to perform this action.',
    404: 'The requested resource was not found.',
    409: 'The request conflicts with the current resource state.',
    422: 'The request could not be processed.',
    429: 'Too many requests. Please try again later.',
  };

  const isServerError = statusCode >= 500;

  if (isServerError) {
    console.error('ASEMS request failed.', {
      method: req.method,
      path: req.path,
      code: error.code,
      message: error.message,
      ...(process.env.NODE_ENV !== 'production' && {
        stack: error.stack,
      }),
    });
  }

  // Errors created with httpError() have a numeric statusCode and a string code,
  // and their messages were written for users (for example "hia is already active.
  // Close it first."). Show those. Parser, library and unknown errors keep the
  // generic text so internal details never leak.
  const isDeliberate =
    !isServerError &&
    Number.isInteger(error.statusCode) &&
    typeof error.code === 'string' &&
    !error.type;

  const message = isServerError
    ? 'An unexpected server error occurred.'
    : isDeliberate
      ? error.message
      : safeClientErrors[statusCode] || error.message;

  return res.status(statusCode).json({
    success: false,
    error: {
      code: isServerError
        ? 'INTERNAL_SERVER_ERROR'
        : error.code || 'REQUEST_ERROR',
      message,
    },
  });
}

      // activate two different cycles at the same moment.

        throw httpError(
          409,
          `A ${current.status} cycle cannot become ${status}.`,
          'INVALID_TRANSITION',
        );
      }

      if (status === 'active') {

          (row) => row.status === 'active' && String(row.id) !== String(id),
        );

          throw httpError(
            409,
            `"${other.title}" is already active. Close it first.`,
            'ANOTHER_CYCLE_ACTIVE',
          );
        }
      }

      await connection.execute('UPDATE evaluation_cycles SET status = ? WHERE id = ?', [
        status,
        id,
      ]);

      await writeAudit(connection, {
        userId: req.user.id,
        action: 'cycle.status_changed',
        entityType: 'evaluation_cycle',
        entityId: current.id,
        details: { previousStatus: current.status, newStatus: status },
        ipAddress: req.ip,
      });

      const [rows] = await connection.execute(`${cycleSelect} WHERE c.id = ?`, [id]);
      return publicCycle(rows[0]);
    });

  }),

async function recomputeInstructorSummary(connection, cycleId, instructorId) {
  const [rows] = await connection.execute(
    `SELECT f.evaluator_type, f.weight, AVG(e.overall_average) AS component_average
       FROM evaluations e
       JOIN evaluation_assignments a ON a.id = e.assignment_id
       JOIN evaluation_forms f ON f.id = a.form_id
      WHERE a.cycle_id = ? AND a.instructor_id = ? AND e.status IN ('submitted', 'approved')
      GROUP BY f.evaluator_type, f.weight`,
    [cycleId, instructorId]
  );
  const byType = new Map(rows.map(row => [row.evaluator_type, {
    average: Number(Number(row.component_average).toFixed(3)),
    weighted: Number((Number(row.component_average) * Number(row.weight) / 5).toFixed(3)),
  }]));
  const required = ['student', 'department_head', 'peer', 'self'];
  const complete = required.every(type => byType.has(type));
  const student = byType.get('student');
  const head = byType.get('department_head');
  const peer = byType.get('peer');
  const self = byType.get('self');
  const total = complete ? required.reduce((sum, type) => sum + byType.get(type).weighted, 0) : 0;

  await connection.execute(
    `INSERT INTO evaluation_summaries
      (cycle_id, instructor_id, student_average, student_weighted,
       department_head_average, department_head_weighted, peer_average, peer_weighted,
       self_average, self_weighted, total_score, status, finalized_by, finalized_at)
     VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, NULL, NULL)
     ON DUPLICATE KEY UPDATE
       student_average = VALUES(student_average), student_weighted = VALUES(student_weighted),
       department_head_average = VALUES(department_head_average), department_head_weighted = VALUES(department_head_weighted),
       peer_average = VALUES(peer_average), peer_weighted = VALUES(peer_weighted),
       self_average = VALUES(self_average), self_weighted = VALUES(self_weighted),
       total_score = VALUES(total_score), status = VALUES(status), finalized_by = NULL, finalized_at = NULL`,
    [cycleId, instructorId,
      student?.average ?? null, student?.weighted ?? 0,
      head?.average ?? null, head?.weighted ?? 0,
      peer?.average ?? null, peer?.weighted ?? 0,
      self?.average ?? null, self?.weighted ?? 0,
      Number(total.toFixed(3)), complete ? 'complete' : 'incomplete']
  );
}

}));

    if (missing.length) throw httpError(400, `Complete every criterion before submitting. ${missing.length} response(s) are missing.`, 'INCOMPLETE_EVALUATION');

    // NA responses are excluded from the mean. This is a provisional technical rule and must be confirmed institutionally.
    const numeric = scoreRows.filter(row => !row.is_na && row.score !== null).map(row => Number(row.score));
    if (!numeric.length) throw httpError(400, 'At least one numeric score is required.', 'NO_NUMERIC_SCORES');
    const average = numeric.reduce((sum, score) => sum + score, 0) / numeric.length;
    const weighted = average * Number(assignment.weight) / 5;

    await connection.execute(
      `UPDATE evaluations SET status = 'submitted', comment_strengths = ?, comment_improvements = ?,
       overall_average = ?, weighted_score = ?, submitted_at = NOW() WHERE id = ?`,
      [parsed.data.commentStrengths ?? null, parsed.data.commentImprovements ?? null, average.toFixed(3), weighted.toFixed(3), evaluation.id]
    );
    await connection.execute("UPDATE evaluation_assignments SET status = 'submitted' WHERE id = ?", [assignmentId]);
    await recomputeInstructorSummary(connection, assignment.cycle_id, assignment.instructor_id);
    await writeAudit(connection, { userId: Number(req.user.id), action: 'evaluation_submitted', entityType: 'evaluation', entityId: evaluation.id, details: { assignmentId, average: Number(average.toFixed(3)), weightedScore: Number(weighted.toFixed(3)) }, ipAddress: req.ip });
    await connection.commit();
    sendData(res, { evaluationId: evaluation.id, assignmentId, status: 'submitted', overallAverage: Number(average.toFixed(3)), weightedScore: Number(weighted.toFixed(3)), note: 'Component scores are recorded; the final instructor score is calculated only when all four component evaluations are submitted.' });
  } catch (error) {
    await connection.rollback();
    throw error;
  } finally {
    connection.release();
  }
}));

function parseId(value, label) {
  if (!/^[1-9]\d*$/.test(String(value))) {
    throw httpError(400, `Invalid ${label}.`, 'VALIDATION_ERROR');

async function inTransaction(work) {
  const connection = await pool.getConnection();
  try {
    await connection.beginTransaction();
    const result = await work(connection);
    await connection.commit();
    return result;
  } catch (error) {
    await connection.rollback();
    throw error;
  } finally {
    connection.release();
  }
}

function audit(connection, req, action, versionId, details = {}) {

    userId: req.user.id,

// Loads one version with its sections and active criteria.
async function loadVersion(executor, versionId) {
  const [versions] = await executor.execute(
    `SELECT v.id, v.form_id, v.version_no, v.status, v.instructions, v.scale_min, v.scale_max,
            v.allow_na, v.scale_labels, v.published_at, v.created_at,
            f.code, f.title AS form_title, f.evaluator_type, f.weight
       FROM form_versions v
       JOIN evaluation_forms f ON f.id = v.form_id
      WHERE v.id = ?`,
    [versionId],
  );
  const version = versions[0];
  if (!version) return null;

  const [sections] = await executor.execute(
    `SELECT id, section_key, title, sort_order
       FROM form_sections WHERE form_version_id = ?
      ORDER BY sort_order, id`,
    [versionId],
  );
  const [criteria] = await executor.execute(
    `SELECT c.id, c.section_id, c.text, c.sort_order
       FROM form_criteria c
       JOIN form_sections s ON s.id = c.section_id
      WHERE s.form_version_id = ? AND c.is_active = 1
      ORDER BY s.sort_order, s.id, c.sort_order, c.id`,
    [versionId],
  );

  return {
    id: Number(version.id),
    formId: Number(version.form_id),
    code: version.code,
    title: version.form_title,
    evaluatorType: version.evaluator_type,
    weight: Number(version.weight),
    versionNo: version.version_no,
    status: version.status,
    instructions: version.instructions,
    allowNa: Number(version.allow_na) === 1,
    scaleMin: version.scale_min,
    scaleMax: version.scale_max,
    scaleLabels: version.scale_labels ?? null,
    ratingScale: buildRatingScale({
      scaleMin: version.scale_min,
      scaleMax: version.scale_max,
      allowNa: Number(version.allow_na) === 1,
    }),
    publishedAt: version.published_at,
    sections: sections.map((section) => ({
      id: Number(section.id),
      sectionKey: section.section_key,
      title: section.title,
      sortOrder: section.sort_order,
      criteria: criteria
        .filter((item) => String(item.section_id) === String(section.id))
        .map((item) => ({ id: Number(item.id), text: item.text, sortOrder: item.sort_order })),
    })),
  };
}

async function insertSections(connection, formId, versionId, sections) {
  for (const [sectionIndex, section] of sections.entries()) {
    const [result] = await connection.execute(
      `INSERT INTO form_sections (form_id, form_version_id, section_key, title, sort_order)
       VALUES (?, ?, ?, ?, ?)`,
      [formId, versionId, section.sectionKey, section.title, sectionIndex + 1],
    );
    for (const [criterionIndex, criterion] of section.criteria.entries()) {
      await connection.execute(
        'INSERT INTO form_criteria (section_id, text, sort_order, is_active) VALUES (?, ?, ?, 1)',
        [result.insertId, criterion.text, criterionIndex + 1],
      );
    }
  }
}

async function lockDraft(connection, versionId) {
  const [rows] = await connection.execute(
    'SELECT id, form_id, status FROM form_versions WHERE id = ? FOR UPDATE',
    [versionId],
  );
  const version = rows[0];
  if (!version) throw httpError(404, 'Form version not found.', 'NOT_FOUND');
  if (version.status !== 'draft') {
    throw httpError(409, 'Only draft versions can be changed. Create a new draft instead.', 'VERSION_LOCKED');
  }
  const [used] = await connection.execute(
    'SELECT COUNT(*) AS count FROM evaluation_assignments WHERE form_version_id = ?',
    [versionId],
  );
  if (Number(used[0].count) > 0) {
    throw httpError(409, 'This version is already used by assignments.', 'VERSION_IN_USE');
  }
  return version;
}

import { Router } from 'express';

// approved through the whole workflow (summary status = 'reviewed').

const SHARE_COMMENTS = true;
const TYPES = ['student', 'department_head', 'peer', 'self'];

const num = (value) => (value === null || value === undefined ? null : Number(value));

// GET /api/results/me
router.get('/me', asyncHandler(async (req, res) => {
  const instructorId = Number(req.user.id);

  const [progressRows] = await pool.execute(
    `SELECT a.cycle_id, cy.title AS cycle_title, cy.academic_year, cy.status AS cycle_status,
            f.evaluator_type,
            COUNT(*) AS total,
            COALESCE(SUM(e.status IN ('submitted', 'approved')), 0) AS submitted,
            COALESCE(SUM(e.status = 'approved'), 0) AS approved
       FROM evaluation_assignments a
       JOIN evaluation_cycles cy ON cy.id = a.cycle_id
       JOIN evaluation_forms f ON f.id = a.form_id
       LEFT JOIN evaluations e ON e.assignment_id = a.id
      WHERE a.instructor_id = ?
      GROUP BY a.cycle_id, cy.title, cy.academic_year, cy.status, cy.start_date, f.evaluator_type
      ORDER BY cy.start_date DESC`,
    [instructorId],
  );

  const [weightRows] = await pool.execute(
    'SELECT evaluator_type, weight FROM evaluation_forms WHERE is_active = 1',
  );
  const weights = Object.fromEntries(weightRows.map((row) => [row.evaluator_type, Number(row.weight)]));

  const [summaryRows] = await pool.execute(
    `SELECT cycle_id, student_average, student_weighted, department_head_average,
            department_head_weighted, peer_average, peer_weighted, self_average,
            self_weighted, total_score, status, finalized_at
       FROM evaluation_summaries WHERE instructor_id = ?`,
    [instructorId],
  );
  const summaries = new Map(summaryRows.map((row) => [Number(row.cycle_id), row]));

  const commentsByCycle = new Map();
  if (SHARE_COMMENTS) {
    // RAND() so the order cannot reveal who submitted first.
    const [commentRows] = await pool.execute(
      `SELECT a.cycle_id, f.evaluator_type, e.comment_strengths, e.comment_improvements
         FROM evaluations e
         JOIN evaluation_assignments a ON a.id = e.assignment_id
         JOIN evaluation_forms f ON f.id = a.form_id
        WHERE a.instructor_id = ? AND e.status = 'approved'
        ORDER BY RAND()`,
      [instructorId],
    );
    commentRows.forEach((row) => {
      const key = Number(row.cycle_id);
      if (!commentsByCycle.has(key)) commentsByCycle.set(key, { strengths: [], improvements: [] });
      const bucket = commentsByCycle.get(key);
      if (row.comment_strengths?.trim()) bucket.strengths.push({ type: row.evaluator_type, text: row.comment_strengths.trim() });
      if (row.comment_improvements?.trim()) bucket.improvements.push({ type: row.evaluator_type, text: row.comment_improvements.trim() });
    });
  }

  const cycles = new Map();
  progressRows.forEach((row) => {
    const id = Number(row.cycle_id);
    if (!cycles.has(id)) {
      cycles.set(id, {
        cycleId: id,
        cycleTitle: row.cycle_title,
        academicYear: row.academic_year,
        cycleStatus: row.cycle_status,
        progress: {},
      });
    }
    cycles.get(id).progress[row.evaluator_type] = {
      total: Number(row.total),
      submitted: Number(row.submitted),
      approved: Number(row.approved),
    };
  });

  const results = [...cycles.values()].map((cycle) => {
    const summary = summaries.get(cycle.cycleId);
    const released = summary?.status === 'reviewed';
    const base = { ...cycle, released, scores: null, comments: null };
    if (!released) return base;

    const components = {};
    TYPES.forEach((type) => {
      components[type] = {
        average: num(summary[`${type}_average`]),
        weighted: num(summary[`${type}_weighted`]),
        weight: weights[type] ?? null,
      };
    });

    return {
      ...base,
      scores: {
        components,
        total: num(summary.total_score),
        maxTotal: TYPES.reduce((sum, type) => sum + (weights[type] ?? 0), 0),
        finalizedAt: summary.finalized_at,
      },
      comments: SHARE_COMMENTS ? commentsByCycle.get(cycle.cycleId) ?? { strengths: [], improvements: [] } : null,
    };
  });

  sendData(res, { results });
}));

router.post('/', asyncHandler(async (req, res) => {
  const parsed = createSchema.safeParse(req.body ?? {});
  if (!parsed.success) {
    const issue = parsed.error.issues[0];
    throw httpError(400, `${issue.path.join('.') || 'body'}: ${issue.message}`, 'VALIDATION_ERROR');
  }
  const input = parsed.data;
  const studentIds = [...new Set(input.studentIds)];
  const peerIds = [...new Set(input.peerIds)];

  if (peerIds.includes(input.instructorId)) {
    throw httpError(400, 'The instructor cannot be their own peer evaluator. Use self evaluation.', 'VALIDATION_ERROR');
  }
  if (!studentIds.length && !input.departmentHeadId && !peerIds.length && !input.includeSelf) {
    throw httpError(400, 'Choose at least one evaluator.', 'VALIDATION_ERROR');
  }

  const connection = await pool.getConnection();
  try {
    await connection.beginTransaction();

    const [cycles] = await connection.execute(
      'SELECT id, status FROM evaluation_cycles WHERE id = ? FOR UPDATE', [input.cycleId],
    );
    if (!cycles[0]) throw httpError(404, 'Evaluation cycle not found.', 'NOT_FOUND');
    if (cycles[0].status === 'closed') throw httpError(409, 'This cycle is closed.', 'CYCLE_CLOSED');

    // Every (user, role) pair must match an active user. A user may fill only the role they really have.
    const wanted = [
      [input.instructorId, 'teacher'],
      ...studentIds.map((userId) => [userId, 'student']),
      ...peerIds.map((userId) => [userId, 'teacher']),
    ];
    if (input.departmentHeadId) wanted.push([input.departmentHeadId, 'department_head']);
    const userIds = [...new Set(wanted.map(([userId]) => userId))];
    const [users] = await connection.query(
      'SELECT id, role, status FROM users WHERE id IN (?)', [userIds],
    );
    const byId = new Map(users.map((user) => [Number(user.id), user]));
    for (const [userId, role] of wanted) {
      const user = byId.get(userId);
      if (!user || user.status !== 'active' || user.role !== role) {
        throw httpError(400, `User ${userId} must be an active ${role}.`, 'INVALID_EVALUATOR');
      }
    }

    // Each task uses the form version that is published right now.
    const [versions] = await connection.execute(
      `SELECT v.id AS version_id, f.id AS form_id, f.evaluator_type
         FROM form_versions v JOIN evaluation_forms f ON f.id = v.form_id
        WHERE v.status = 'published' AND f.is_active = 1`,
    );
    const formFor = new Map(versions.map((row) => [row.evaluator_type, row]));

    const planned = [
      ...studentIds.map((evaluatorId) => ['student', evaluatorId]),
      ...(input.departmentHeadId ? [['department_head', input.departmentHeadId]] : []),
      ...peerIds.map((evaluatorId) => ['peer', evaluatorId]),
      ...(input.includeSelf ? [['self', input.instructorId]] : []),
    ];
    for (const [type] of planned) {
      if (!formFor.has(type)) throw httpError(409, `No published form exists for ${type} evaluation.`, 'FORM_NOT_PUBLISHED');
    }

    const [existing] = await connection.execute(
      `SELECT form_id, evaluator_id FROM evaluation_assignments
        WHERE cycle_id = ? AND instructor_id = ? AND course_code = ?`,
      [input.cycleId, input.instructorId, input.courseCode],
    );
    const exists = new Set(existing.map((row) => `${row.form_id}:${row.evaluator_id}`));

    let created = 0;
    let skipped = 0;
    for (const [type, evaluatorId] of planned) {
      const form = formFor.get(type);
      if (exists.has(`${form.form_id}:${evaluatorId}`)) { skipped += 1; continue; }
      await connection.execute(
        `INSERT INTO evaluation_assignments
           (cycle_id, form_id, form_version_id, instructor_id, evaluator_id,
            course_title, course_code, batch, semester, status)
         VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, 'pending')`,
        [input.cycleId, form.form_id, form.version_id, input.instructorId, evaluatorId,
          input.courseTitle, input.courseCode, input.batch ?? null, input.semester ?? null],
      );
      created += 1;
    }

    await writeAudit(connection, {
      userId: req.user.id,
      action: 'tasks.created',
      entityType: 'evaluation_cycle',
      entityId: input.cycleId,
      details: { instructorId: input.instructorId, courseCode: input.courseCode, created, skipped },
      ipAddress: req.ip,
    });
    await connection.commit();
    sendData(res, { created, skipped }, 201);
  } catch (error) {
    await connection.rollback();
    throw error;
  } finally {
    connection.release();
  }
}));

let server;

async function startServer() {
  try {
    const database = await verifyDatabaseConnection();

    console.info('ASEMS database connection verified.', {
      database: database.database,
    });

    server = app.listen(env.PORT, () => {
      console.info('ASEMS API is ready.', {
        port: env.PORT,
        environment: env.NODE_ENV,
      });
    });

    server.on('error', (error) => {
      console.error('ASEMS HTTP server error.', {
        code: error.code,
        message: error.message,
      });

      void shutdown('SERVER_ERROR', 1);
    });
  } catch (error) {
    console.error('ASEMS startup failed.', {
      message: error.message,
      code: error.code,
    });

    await closeDatabaseConnection().catch(() => {});
    process.exitCode = 1;
  }
}

async function shutdown(signal, exitCode = 0) {
  if (isShuttingDown) return;
  isShuttingDown = true;

  console.info(`ASEMS shutdown started: ${signal}`);

  const forceExitTimer = setTimeout(() => {
    console.error('Graceful shutdown timed out.');
    process.exit(1);
  }, 10_000);

  forceExitTimer.unref();

  try {
    if (server) {
      await new Promise((resolve, reject) => {
        server.close((error) => {
          if (error) reject(error);
          else resolve();
        });
      });
    }

    await closeDatabaseConnection();
  } catch (error) {
    console.error('ASEMS shutdown encountered an error.', {
      message: error.message,
    });

    exitCode = 1;
  } finally {
    clearTimeout(forceExitTimer);
    process.exitCode = exitCode;
  }
}

process.once('SIGINT', () => {
  void shutdown('SIGINT');
});

process.once('SIGTERM', () => {
  void shutdown('SIGTERM');
});

void startServer();

    action: 'auth.login',

  const status = query.status || '';

  if (role && !ROLES.includes(role)) fail('Invalid role filter.');
  if (status && !STATUSES.includes(status)) fail('Invalid status filter.');

  const conditions = [];
  const params = [];

  if (search) {
    conditions.push('(name LIKE ? OR email LIKE ? OR id_number LIKE ?)');
    const term = `%${search}%`;
    params.push(term, term, term);
  }

  if (role) {
    conditions.push('role = ?');
    params.push(role);
  }

  if (status) {
    conditions.push('status = ?');
    params.push(status);
  }

  const where = conditions.length
    ? `WHERE ${conditions.join(' AND ')}`
    : '';

  const [countRows] = await pool.execute(
    `SELECT COUNT(*) AS total FROM users ${where}`,
    params,
  );

  const [rows] = await pool.execute(
    `SELECT ${publicFields}
       FROM users
       ${where}
      ORDER BY created_at DESC, id DESC
      LIMIT ? OFFSET ?`,
    [...params, limit, offset],
  );

  return {
    users: rows.map(publicUser),
    pagination: {
      page,
      limit,
      total: Number(countRows[0].total),
      totalPages: Math.ceil(Number(countRows[0].total) / limit),
    },
  };
}

export async function createUser(input, req) {
  const data = nullableFields(parseUser(input));

  if (
    typeof input.password !== 'string' ||
    input.password.length < 12 ||
    input.password.length > 128
  ) {
    fail('Initial password must be between 12 and 128 characters.');
  }

  const passwordHash = await bcrypt.hash(input.password, 12);
  const connection = await pool.getConnection();

  try {
    await connection.beginTransaction();

    const [result] = await connection.execute(
      `INSERT INTO users
        (name, email, password, role, status, phone, gender,
         date_of_birth, address, id_number, profile_photo)
       VALUES (?, ?, ?, ?, 'active', ?, ?, ?, ?, ?, ?)`,
      [
        data.name,
        data.email,
        passwordHash,
        data.role,
        data.phone ?? null,
        data.gender ?? null,
        data.date_of_birth ?? null,
        data.address ?? null,
        data.id_number ?? null,
        data.profile_photo ?? null,
      ],
    );

    await audit(connection, req, 'user.created', result.insertId, {
      role: data.role,
    });

    const [rows] = await connection.execute(
      `SELECT ${publicFields} FROM users WHERE id = ?`,
      [result.insertId],
    );

    await connection.commit();

    return publicUser(rows[0]);
  } catch (error) {
    await connection.rollback();

    if (isDuplicateEmail(error)) {
      fail('A user with this email or ID number already exists.', 409, 'DUPLICATE_USER');
    }

    throw error;
  } finally {
    connection.release();
  }
}

export async function updateUser(id, input, req) {
  validateId(id);

  const data = nullableFields(parseUser(input));
  const connection = await pool.getConnection();

  try {
    await connection.beginTransaction();

    const [existingRows] = await connection.execute(
      `SELECT id, role, status FROM users WHERE id = ? FOR UPDATE`,
      [id],
    );

    const existing = existingRows[0];
    if (!existing) fail('User not found.', 404, 'USER_NOT_FOUND');

    if (
      existing.role === 'admin' &&
      existing.status === 'active' &&
      data.role !== 'admin'
    ) {
      const [admins] = await connection.execute(
        `SELECT id FROM users
         WHERE role = 'admin' AND status = 'active'
         FOR UPDATE`,
      );

      if (admins.length <= 1) {
        fail('Cannot demote the last active administrator.', 409, 'LAST_ADMIN');
      }
    }

    await connection.execute(
      `UPDATE users
          SET name = ?, email = ?, role = ?, phone = ?, gender = ?,
              date_of_birth = ?, address = ?, id_number = ?, profile_photo = ?
        WHERE id = ?`,
      [
        data.name,
        data.email,
        data.role,
        data.phone ?? null,
        data.gender ?? null,
        data.date_of_birth ?? null,
        data.address ?? null,
        data.id_number ?? null,
        data.profile_photo ?? null,
        id,
      ],
    );

    await audit(connection, req, 'user.updated', id, {
      previousRole: existing.role,
      newRole: data.role,
    });

    const [rows] = await connection.execute(
      `SELECT ${publicFields} FROM users WHERE id = ?`,
      [id],
    );

    await connection.commit();

    return publicUser(rows[0]);
  } catch (error) {
    await connection.rollback();

    if (isDuplicateEmail(error)) {
      fail('A user with this email or ID number already exists.', 409, 'DUPLICATE_USER');
    }

    throw error;
  } finally {
    connection.release();
  }
}

export async function changeUserStatus(id, status, req) {
  validateId(id);

  if (!STATUSES.includes(status)) fail('Invalid account status.');

  const connection = await pool.getConnection();

  try {
    await connection.beginTransaction();

    const [rows] = await connection.execute(
      `SELECT id, role, status FROM users WHERE id = ? FOR UPDATE`,
      [id],
    );

    const user = rows[0];
    if (!user) fail('User not found.', 404, 'USER_NOT_FOUND');

    if (Number(user.id) === Number(req.user.id) && status !== 'active') {
      fail('You cannot deactivate or suspend your own account.', 409, 'SELF_DEACTIVATION');
    }

    if (
      user.role === 'admin' &&
      user.status === 'active' &&
      status !== 'active'
    ) {
      const [admins] = await connection.execute(
        `SELECT id FROM users
         WHERE role = 'admin' AND status = 'active'
         FOR UPDATE`,
      );

      if (admins.length <= 1) {
        fail('Cannot deactivate the last active administrator.', 409, 'LAST_ADMIN');
      }
    }

    await connection.execute(
      'UPDATE users SET status = ? WHERE id = ?',
      [status, id],
    );

    await audit(connection, req, 'user.status_changed', id, {
      previousStatus: user.status,
      newStatus: status,
    });

    await connection.commit();

    return { id: user.id, status };
  } catch (error) {
    await connection.rollback();
    throw error;
  } finally {
    connection.release();
  }
}

// Shape of the editable content of a DRAFT form version.
export const draftContentSchema = z.object({
  instructions: z.string().trim().max(5000).nullable().optional(),
  allowNa: z.boolean().optional(),
  sections: z
    .array(
      z.object({
        sectionKey: z
          .string()
          .trim()
          .min(1)
          .max(50)
          .regex(/^[A-Za-z0-9_-]+$/, 'Section keys may only contain letters, digits, "_" and "-".'),
        title: z.string().trim().min(1).max(255),
        criteria: z
          .array(z.object({ text: z.string().trim().min(1).max(600) }))
          .min(1)
          .max(100),
      }),
    )
    .min(1)
    .max(30),
});

// Rules that Zod cannot express. Returns a list of human-readable problems (empty when valid).
export function findContentProblems(sections) {
  const problems = [];
  const keys = new Set();

  sections.forEach((section, index) => {
    const key = section.sectionKey.toLowerCase();
    if (keys.has(key)) problems.push(`Section key "${section.sectionKey}" is used more than once.`);
    keys.add(key);
    if (section.criteria.length === 0) problems.push(`Section ${index + 1} has no criteria.`);
  });

  return problems;
}

// Official forms use 5, 4, 3, 2, 1, 0 and NA.
export function buildRatingScale({ scaleMin, scaleMax, allowNa }) {
  const scale = [];
  for (let value = scaleMax; value >= scaleMin; value -= 1) scale.push(value);
  if (allowNa) scale.push('NA');
  return scale;
}

// Returns an error message when the score is not allowed on this version, otherwise null.
export function scoreProblem(score, { scale_min: min, scale_max: max, allow_na: allowNa }) {
  if (score === 'NA') return Number(allowNa) === 1 ? null : 'NA is not permitted on this form.';
  if (score < Number(min) || score > Number(max)) {
    return `Scores must be between ${Number(min)} and ${Number(max)}.`;
  }
  return null;
}
